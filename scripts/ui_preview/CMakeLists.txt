cmake_minimum_required(VERSION 3.16)
project(mic_on_preview C CXX)
get_filename_component(FW "${CMAKE_CURRENT_SOURCE_DIR}/../../firmware" ABSOLUTE)
set(LVGL_DIR "${FW}/managed_components/lvgl__lvgl" CACHE PATH "Matching LVGL source tree")
set(LV_BUILD_CONF_PATH ${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_FETCH_DEPENDENCIES OFF CACHE BOOL "" FORCE)
add_subdirectory(${LVGL_DIR} lvgl)
add_executable(preview preview.c ${FW}/main/fonts/ui_font_14.c ${FW}/main/fonts/ui_font_16.c ${FW}/main/fonts/ui_font_18.c ${FW}/main/fonts/ui_font_22.c ${FW}/main/assets/ui_wordmark.c ${FW}/main/fonts/ui_font_comic_80.c ${FW}/main/fonts/ui_font_comic_52.c ${FW}/main/fonts/ui_font_comic_44.c ${FW}/main/menu.c)
# The UI is included through a macro; older CMake scanners can miss this dependency.
set_source_files_properties(preview.c PROPERTIES OBJECT_DEPENDS "${FW}/main/ui.c")
target_compile_definitions(preview PRIVATE PREVIEW_UI_SOURCE="${FW}/main/ui.c")
target_include_directories(preview PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${FW}/main)
target_link_libraries(preview PRIVATE lvgl m)
